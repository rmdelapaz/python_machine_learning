Introduction

Quantum Machine Learning (QML) combines quantum computing with machine learning algorithms, potentially offering exponential speedups for certain problems. By leveraging quantum phenomena like superposition and entanglement, QML can process information in fundamentally different ways than classical computers. This lesson explores quantum computing basics, quantum algorithms for ML, and practical implementations using quantum simulators.

Quantum State Simulation

class QuantumSimulator:
    """
    Simple quantum state simulator for educational purposes
    """
    
    def __init__(self):
        self.state = None
        self.n_qubits = 0
        
    def initialize_state(self, n_qubits):
        """
        Initialize quantum state to |00...0⟩
        """
        self.n_qubits = n_qubits
        self.state = np.zeros(2**n_qubits, dtype=complex)
        self.state[0] = 1.0  # |00...0⟩ state
        
        print(f"Initialized {n_qubits}-qubit state")
        print(f"State vector dimension: {2**n_qubits}")
        
    def apply_hadamard(self, qubit):
        """
        Apply Hadamard gate to create superposition
        """
        H = np.array([[1, 1], [1, -1]]) / np.sqrt(2)
        self._apply_single_qubit_gate(H, qubit)
        print(f"Applied Hadamard to qubit {qubit}")
        
    def apply_cnot(self, control, target):
        """
        Apply CNOT (controlled-NOT) gate
        """
        n = 2**self.n_qubits
        new_state = self.state.copy()
        
        for i in range(n):
            if (i >> control) & 1:  # Control qubit is |1⟩
                # Flip target qubit
                flipped = i ^ (1 << target)
                new_state[flipped] = self.state[i]
                new_state[i] = self.state[flipped]
        
        self.state = new_state
        print(f"Applied CNOT: control={control}, target={target}")
        
    def _apply_single_qubit_gate(self, gate, qubit):
        """
        Apply single-qubit gate to quantum state
        """
        n = 2**self.n_qubits
        new_state = np.zeros(n, dtype=complex)
        
        for i in range(n):
            # Get bit value at qubit position
            bit = (i >> qubit) & 1
            
            # Calculate new amplitudes
            for new_bit in range(2):
                # Flip bit at qubit position
                j = i ^ ((bit ^ new_bit) << qubit)
                new_state[j] += gate[new_bit, bit] * self.state[i]
        
        self.state = new_state
        
    def measure(self, n_shots=1000):
        """
        Simulate measurement of quantum state
        """
        probabilities = np.abs(self.state)**2
        
        # Sample from probability distribution
        outcomes = np.random.choice(
            len(probabilities),
            size=n_shots,
            p=probabilities
        )
        
        # Count occurrences
        counts = {}
        for outcome in outcomes:
            # Convert to binary string
            binary = format(outcome, f'0{self.n_qubits}b')
            counts[binary] = counts.get(binary, 0) + 1
        
        return counts
    
    def visualize_state(self):
        """
        Visualize quantum state
        """
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        # Probability distribution
        probabilities = np.abs(self.state)**2
        labels = [format(i, f'0{self.n_qubits}b') for i in range(len(self.state))]
        
        axes[0].bar(labels, probabilities)
        axes[0].set_xlabel('Basis State')
        axes[0].set_ylabel('Probability')
        axes[0].set_title('Quantum State Probabilities')
        axes[0].tick_params(axis='x', rotation=45)
        
        # Phase visualization
        phases = np.angle(self.state)
        axes[1].bar(labels, phases)
        axes[1].set_xlabel('Basis State')
        axes[1].set_ylabel('Phase (radians)')
        axes[1].set_title('Quantum State Phases')
        axes[1].tick_params(axis='x', rotation=45)
        
        plt.suptitle('Quantum State Visualization', fontsize=14)
        plt.tight_layout()
        plt.show()

class QuantumMLAlgorithms:
    """
    Quantum-inspired machine learning algorithms
    """
    
    def __init__(self):

    def quantum_kernel(self, X, gamma=1.0):
        """
        Quantum-inspired kernel function
        """
        print("\n" + "=" * 60)
        print("QUANTUM KERNEL METHOD")
        print("=" * 60)
        
        n_samples = len(X)
        
        # Classical RBF kernel for comparison
        classical_kernel = np.zeros((n_samples, n_samples))
        for i in range(n_samples):
            for j in range(n_samples):
                diff = X[i] - X[j]
                classical_kernel[i, j] = np.exp(-gamma * np.dot(diff, diff))
        
        # Quantum-inspired kernel (using feature map)
        print("Computing quantum feature map...")
        
        # Simulate quantum feature mapping
        def quantum_feature_map(x):
            """Map classical data to quantum state"""
            # Simplified: use trigonometric encoding
            phi = np.array([
                np.cos(np.pi * x[0]),
                np.sin(np.pi * x[0]),
                np.cos(np.pi * x[1]),
                np.sin(np.pi * x[1])
            ])
            return phi / np.linalg.norm(phi)
        
        # Compute quantum kernel
        quantum_kernel = np.zeros((n_samples, n_samples))
        for i in range(n_samples):
            phi_i = quantum_feature_map(X[i])
            for j in range(n_samples):
                phi_j = quantum_feature_map(X[j])
                # Inner product in feature space
                quantum_kernel[i, j] = np.abs(np.dot(phi_i, phi_j))**2
        
        print(f"Kernel matrix shape: {quantum_kernel.shape}")
        
        # Visualize kernels
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        im1 = axes[0].imshow(classical_kernel, cmap='viridis')
        axes[0].set_title('Classical RBF Kernel')
        axes[0].set_xlabel('Sample')
        axes[0].set_ylabel('Sample')
        plt.colorbar(im1, ax=axes[0])
        
        im2 = axes[1].imshow(quantum_kernel, cmap='viridis')
        axes[1].set_title('Quantum-Inspired Kernel')
        axes[1].set_xlabel('Sample')
        axes[1].set_ylabel('Sample')
        plt.colorbar(im2, ax=axes[1])
        
        plt.suptitle('Kernel Matrix Comparison', fontsize=14)
        plt.tight_layout()
        plt.show()
        
        return quantum_kernel
    
    def variational_quantum_classifier(self, X, y, n_qubits=2, n_layers=2):
        """
        Simplified Variational Quantum Classifier (VQC)
        """
        print("\n" + "=" * 60)
        print("VARIATIONAL QUANTUM CLASSIFIER")
        print("=" * 60)
        
        n_samples = len(X)
        
        # Initialize parameters
        n_params = n_qubits * n_layers * 3  # 3 rotation gates per qubit per layer
        params = np.random.randn(n_params) * 0.1
        
        def quantum_circuit(x, theta):
            """
            Parameterized quantum circuit
            """
            # Encode classical data
            encoded = np.array([np.cos(np.pi * x[0]), np.sin(np.pi * x[1])])
            
            # Apply variational layers (simplified)
            state = encoded
            for layer in range(n_layers):
                # Rotation gates
                angle = theta[layer * 3]
                rotation = np.array([
                    [np.cos(angle), -np.sin(angle)],
                    [np.sin(angle), np.cos(angle)]
                ])
                state = rotation @ state
            
            # Measurement (expectation value)
            return state[0]**2  # Probability of measuring |0⟩
        
        # Training loop (simplified gradient-free optimization)
        learning_rate = 0.1
        n_epochs = 20
        losses = []
        
        print(f"Training VQC with {n_params} parameters...")
        print(f"Circuit depth: {n_layers} layers")
        
        for epoch in range(n_epochs):
            # Forward pass
            predictions = []
            for x in X:
                pred = quantum_circuit(x, params)
                predictions.append(pred)
            
            # Compute loss (MSE)
            predictions = np.array(predictions)
            loss = np.mean((predictions - y)**2)
            losses.append(loss)
            
            # Parameter update (simplified gradient estimation)
            for i in range(n_params):
                # Finite difference approximation
                epsilon = 0.01
                params_plus = params.copy()
                params_plus[i] += epsilon
                
                # Compute gradient
                preds_plus = [quantum_circuit(x, params_plus) for x in X]
                loss_plus = np.mean((np.array(preds_plus) - y)**2)
                
                gradient = (loss_plus - loss) / epsilon
                params[i] -= learning_rate * gradient
            
            if epoch % 5 == 0:
                print(f"  Epoch {epoch}: Loss = {loss:.4f}")
        
        # Visualize training
        plt.figure(figsize=(10, 5))
        plt.subplot(1, 2, 1)
        plt.plot(losses)
        plt.xlabel('Epoch')
        plt.ylabel('Loss')
        plt.title('VQC Training Loss')
        plt.grid(True, alpha=0.3)
        
        # Decision boundary
        plt.subplot(1, 2, 2)
        xx, yy = np.meshgrid(np.linspace(-2, 2, 50), np.linspace(-2, 2, 50))
        Z = np.array([quantum_circuit([x, y], params) 
                     for x, y in zip(xx.ravel(), yy.ravel())])
        Z = Z.reshape(xx.shape)
        
        plt.contourf(xx, yy, Z, alpha=0.3, cmap='coolwarm')
        plt.scatter(X[:, 0], X[:, 1], c=y, cmap='coolwarm', edgecolors='black')
        plt.xlabel('Feature 1')
        plt.ylabel('Feature 2')
        plt.title('VQC Decision Boundary')
        
        plt.suptitle('Variational Quantum Classifier', fontsize=14)
        plt.tight_layout()
        plt.show()
        
        return params, losses
    
    def quantum_pca(self, X, n_components=2):
        """
        Quantum-inspired PCA
        """
        print("\n" + "=" * 60)
        print("QUANTUM PCA")
        print("=" * 60)
        
        # Classical PCA for comparison
        classical_pca = PCA(n_components=n_components)
        X_classical = classical_pca.fit_transform(X)
        
        print("Quantum PCA uses quantum phase estimation")
        print("to find eigenvalues exponentially faster")
        
        # Simplified quantum-inspired approach
        # Use quantum-inspired techniques
        
        # Compute covariance matrix
        X_centered = X - np.mean(X, axis=0)
        cov_matrix = np.cov(X_centered.T)
        
        # Quantum phase estimation (simulated)
        print("\nSimulating quantum phase estimation...")
        
        # In real quantum PCA, this would use quantum circuits
        eigenvalues, eigenvectors = np.linalg.eig(cov_matrix)
        
        # Sort by eigenvalue
        idx = np.argsort(eigenvalues)[::-1]
        eigenvalues = eigenvalues[idx][:n_components]
        eigenvectors = eigenvectors[:, idx][:, :n_components]
        
        # Transform data
        X_quantum = X_centered @ eigenvectors
        
        # Visualize
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        axes[0].scatter(X_classical[:, 0], X_classical[:, 1], alpha=0.6)

        plt.suptitle('PCA Comparison', fontsize=14)

        plt.show()
        
        print(f"Eigenvalues: {eigenvalues}")
        print(f"Explained variance ratio: {eigenvalues / np.sum(eigenvalues)}")
        
        return X_quantum, eigenvalues, eigenvectors

class HybridQuantumClassical:
    """
    Hybrid quantum-classical machine learning
    """
    
    def __init__(self):
        self.quantum_params = None
        self.classical_params = None
        
    def quantum_feature_extraction(self, X):
        """
        Use quantum circuit for feature extraction
        """
        print("\n" + "=" * 60)
        print("QUANTUM FEATURE EXTRACTION")
        print("=" * 60)
        
        n_samples, n_features = X.shape
        
        # Simulate quantum feature extraction
        print("Encoding classical data into quantum states...")
        
        quantum_features = []
        for x in X:
            # Amplitude encoding
            amplitudes = x / np.linalg.norm(x)
            
            # Apply quantum transformations (simplified)
            # In reality, this would involve quantum gates
            transformed = np.array([
                np.cos(np.pi * amplitudes[0]) * np.sin(np.pi * amplitudes[1]),
                np.sin(np.pi * amplitudes[0]) * np.cos(np.pi * amplitudes[1]),
                np.cos(2 * np.pi * np.mean(amplitudes)),
                np.sin(2 * np.pi * np.std(amplitudes))
            ])
            
            quantum_features.append(transformed)
        
        quantum_features = np.array(quantum_features)
        
        print(f"Original features: {n_features}")
        print(f"Quantum features: {quantum_features.shape[1]}")
        
        return quantum_features
    
    def hybrid_neural_network(self, X, y, quantum_layers=2, classical_layers=2):
        """
        Hybrid quantum-classical neural network
        """
        print("\n" + "=" * 60)
        print("HYBRID QUANTUM-CLASSICAL NEURAL NETWORK")
        print("=" * 60)
        
        # Architecture description
        print(f"Architecture:")
        print(f"  Input → {quantum_layers} Quantum Layers → "
              f"{classical_layers} Classical Layers → Output")
        
        # Quantum feature extraction
        X_quantum = self.quantum_feature_extraction(X)
        
        # Classical neural network on quantum features
        from sklearn.neural_network import MLPClassifier
        
        classical_nn = MLPClassifier(
            hidden_layer_sizes=(32, 16),
            activation='relu',
            max_iter=500,
            random_state=42
        )
        
        # Train classical part
        print("\nTraining classical layers on quantum features...")
        classical_nn.fit(X_quantum, y)
        
        # Evaluate
        train_score = classical_nn.score(X_quantum, y)
        print(f"Training accuracy: {train_score:.4f}")
        
        return classical_nn, X_quantum
    
    def quantum_optimization(self, objective_function, n_iterations=50):
        """
        Quantum-inspired optimization (QAOA-like)
        """
        print("\n" + "=" * 60)
        print("QUANTUM OPTIMIZATION (QAOA-INSPIRED)")
        print("=" * 60)
        
        # Initialize parameters
        n_params = 4
        params = np.random.randn(n_params)
        
        # Optimization history
        history = {'params': [], 'values': []}
        
        print("Optimizing using quantum-inspired techniques...")
        
        for iteration in range(n_iterations):
            # Evaluate objective
            value = objective_function(params)
            
            # Store history
            history['params'].append(params.copy())
            history['values'].append(value)
            
            # Quantum-inspired update
            # Use superposition-like exploration
            perturbations = []
            for i in range(2**n_params):
                # Binary representation for perturbation direction
                binary = format(i, f'0{n_params}b')
                perturbation = np.array([1 if b == '1' else -1 for b in binary])
                perturbations.append(perturbation * 0.1)
            
            # Evaluate all perturbations (quantum parallelism)
            perturbed_values = []
            for pert in perturbations:
                perturbed_params = params + pert
                perturbed_values.append(objective_function(perturbed_params))
            
            # Select best direction (measurement)
            best_idx = np.argmin(perturbed_values)
            params = params + perturbations[best_idx]
            
            if iteration % 10 == 0:
                print(f"  Iteration {iteration}: Value = {value:.4f}")
        
        # Visualize optimization
        plt.figure(figsize=(10, 5))
        plt.plot(history['values'])
        plt.xlabel('Iteration')
        plt.ylabel('Objective Value')
        plt.title('Quantum-Inspired Optimization Progress')
        plt.grid(True, alpha=0.3)
        plt.show()
        
        return params, history

def rosenbrock(x):
    """Rosenbrock function for optimization testing"""
    return sum(100 * (x[i+1] - x[i]**2)**2 + (1 - x[i])**2 
               for i in range(len(x) - 1))

🎓 Key Takeaways

  • Qubits exploit superposition and entanglement to represent and process information differently from classical bits.
  • Quantum feature maps encode data into quantum states, and variational circuits act as trainable models.
  • Most practical quantum ML today is hybrid: a classical optimizer tunes the parameters of a quantum circuit.

❓ Common Questions at This Stage

Do I need a real quantum computer to learn this?

No. Simulators like those in Qiskit or PennyLane run quantum circuits on a normal computer, which is the standard way to learn and prototype before touching real (and noisy) hardware.

Will quantum computers replace classical ML soon?

Unlikely in the near term. Current devices are small and noisy, and a clear, broad quantum advantage for ML hasn't been demonstrated. Treat quantum ML as a promising research frontier, not a production default.

What is a variational quantum classifier?

It's a parameterized quantum circuit whose rotation angles are trained — like weights — by a classical optimizer to minimize a loss, forming a hybrid quantum-classical model.
