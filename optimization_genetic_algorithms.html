🧬 Genetic Algorithms: Evolution-Inspired Optimization

Introduction

Genetic Algorithms (GA) are optimization techniques inspired by natural evolution. They work by evolving a population of candidate solutions through selection, crossover, and mutation operations. GAs are particularly effective for complex optimization problems where traditional methods struggle, including feature selection, hyperparameter tuning, neural architecture search, and combinatorial optimization problems.

Basic Genetic Algorithm Implementation

class GeneticAlgorithm:
    """
    Basic Genetic Algorithm implementation
    """
    
    def __init__(self, 
                 population_size: int = 50,
                 chromosome_length: int = 10,
                 generations: int = 100,
                 crossover_rate: float = 0.8,
                 mutation_rate: float = 0.01,
                 elitism: int = 2,
                 selection_method: str = 'tournament',
                 verbose: bool = True):
        """
        Initialize Genetic Algorithm
        
        Parameters:
        -----------
        population_size : int
            Number of individuals in population
        chromosome_length : int
            Length of each chromosome
        generations : int
            Number of generations to evolve
        crossover_rate : float
            Probability of crossover
        mutation_rate : float
            Probability of mutation per gene
        elitism : int
            Number of best individuals to preserve
        selection_method : str
            Selection strategy ('tournament', 'roulette', 'rank')
        """
        self.population_size = population_size
        self.chromosome_length = chromosome_length
        self.generations = generations
        self.crossover_rate = crossover_rate
        self.mutation_rate = mutation_rate
        self.elitism = elitism
        self.selection_method = selection_method
        self.verbose = verbose
        
        self.population = None
        self.best_individual = None
        self.best_fitness = -np.inf
        self.fitness_history = []
        
    def initialize_population(self) -> np.ndarray:
        """
        Initialize random population
        """
        return np.random.randint(0, 2, size=(self.population_size, self.chromosome_length))
    
    def fitness_function(self, individual: np.ndarray) -> float:
        """
        Calculate fitness of an individual (to be overridden)
        """
        # Example: maximize the number of 1s
        return np.sum(individual)
    
    def evaluate_population(self, population: np.ndarray) -> np.ndarray:
        """
        Evaluate fitness for entire population
        """
        return np.array([self.fitness_function(ind) for ind in population])
    
    def selection(self, population: np.ndarray, fitness_scores: np.ndarray) -> np.ndarray:
        """
        Select parents for next generation
        """
        if self.selection_method == 'tournament':
            return self.tournament_selection(population, fitness_scores)
        elif self.selection_method == 'roulette':
            return self.roulette_wheel_selection(population, fitness_scores)
        elif self.selection_method == 'rank':
            return self.rank_selection(population, fitness_scores)
        else:
            raise ValueError(f"Unknown selection method: {self.selection_method}")
    
    def tournament_selection(self, population: np.ndarray, fitness_scores: np.ndarray, 
                            tournament_size: int = 3) -> np.ndarray:
        """
        Tournament selection
        """
        selected_idx = []
        
        for _ in range(len(population)):
            # Random tournament
            tournament_idx = np.random.choice(len(population), tournament_size, replace=False)
            tournament_fitness = fitness_scores[tournament_idx]
            
            # Select winner
            winner_idx = tournament_idx[np.argmax(tournament_fitness)]
            selected_idx.append(winner_idx)
        
        return population[selected_idx]
    
    def roulette_wheel_selection(self, population: np.ndarray, fitness_scores: np.ndarray) -> np.ndarray:
        """
        Roulette wheel selection
        """
        # Ensure positive fitness values
        min_fitness = np.min(fitness_scores)
        if min_fitness < 0:
            fitness_scores = fitness_scores - min_fitness + 1
        
        # Calculate selection probabilities
        total_fitness = np.sum(fitness_scores)
        probabilities = fitness_scores / total_fitness
        
        # Select individuals
        selected_idx = np.random.choice(
            len(population), 
            size=len(population), 
            p=probabilities, 
            replace=True
        )
        
        return population[selected_idx]
    
    def rank_selection(self, population: np.ndarray, fitness_scores: np.ndarray) -> np.ndarray:
        """
        Rank-based selection
        """
        # Rank individuals
        ranks = np.argsort(np.argsort(fitness_scores)) + 1
        
        # Calculate selection probabilities based on rank
        probabilities = ranks / np.sum(ranks)
        
        # Select individuals
        selected_idx = np.random.choice(
            len(population), 
            size=len(population), 
            p=probabilities, 
            replace=True
        )
        
        return population[selected_idx]
    
    def crossover(self, parent1: np.ndarray, parent2: np.ndarray) -> Tuple[np.ndarray, np.ndarray]:
        """
        Single-point crossover
        """
        if np.random.random() < self.crossover_rate:
            # Choose crossover point
            crossover_point = np.random.randint(1, len(parent1))
            
            # Create offspring
            offspring1 = np.concatenate([parent1[:crossover_point], parent2[crossover_point:]])
            offspring2 = np.concatenate([parent2[:crossover_point], parent1[crossover_point:]])
            
            return offspring1, offspring2
        else:
            return parent1.copy(), parent2.copy()
    
    def mutation(self, individual: np.ndarray) -> np.ndarray:
        """
        Bit-flip mutation for binary encoding
        """
        for i in range(len(individual)):
            if np.random.random() < self.mutation_rate:
                individual[i] = 1 - individual[i]  # Flip bit
        
        return individual
    
    def evolve(self):
        """
        Main evolution loop
        """
        # Initialize population
        self.population = self.initialize_population()
        
        for generation in range(self.generations):
            # Evaluate fitness
            fitness_scores = self.evaluate_population(self.population)
            
            # Track best individual
            best_idx = np.argmax(fitness_scores)
            if fitness_scores[best_idx] > self.best_fitness:
                self.best_fitness = fitness_scores[best_idx]
                self.best_individual = self.population[best_idx].copy()
            
            self.fitness_history.append({
                'generation': generation,
                'best_fitness': self.best_fitness,
                'avg_fitness': np.mean(fitness_scores),
                'min_fitness': np.min(fitness_scores)
            })
            
            if self.verbose and generation % 10 == 0:
                print(f"Generation {generation}: Best Fitness = {self.best_fitness:.4f}, "
                      f"Avg = {np.mean(fitness_scores):.4f}")
            
            # Create new population
            new_population = []
            
            # Elitism: preserve best individuals
            if self.elitism > 0:
                elite_idx = np.argsort(fitness_scores)[-self.elitism:]
                for idx in elite_idx:
                    new_population.append(self.population[idx].copy())
            
            # Selection
            selected_population = self.selection(self.population, fitness_scores)
            
            # Crossover and mutation
            while len(new_population) < self.population_size:
                # Select parents
                parent_idx = np.random.choice(len(selected_population), 2, replace=False)
                parent1 = selected_population[parent_idx[0]]
                parent2 = selected_population[parent_idx[1]]
                
                # Crossover
                offspring1, offspring2 = self.crossover(parent1, parent2)
                
                # Mutation
                offspring1 = self.mutation(offspring1)
                offspring2 = self.mutation(offspring2)
                
                # Add to new population
                new_population.append(offspring1)
                if len(new_population) < self.population_size:
                    new_population.append(offspring2)
            
            self.population = np.array(new_population[:self.population_size])
        
        if self.verbose:
            print(f"\nEvolution complete!")
            print(f"Best fitness: {self.best_fitness:.4f}")
            print(f"Best individual: {self.best_individual}")
    
    def plot_evolution(self):
        """
        Plot evolution progress
        """
        df_history = pd.DataFrame(self.fitness_history)
        
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        # Fitness over generations
        axes[0].plot(df_history['generation'], df_history['best_fitness'], 
                    label='Best', linewidth=2, color='green')
        axes[0].plot(df_history['generation'], df_history['avg_fitness'], 
                    label='Average', linewidth=2, color='blue')
        axes[0].plot(df_history['generation'], df_history['min_fitness'], 
                    label='Minimum', linewidth=2, color='red', alpha=0.5)
        axes[0].set_xlabel('Generation')
        axes[0].set_ylabel('Fitness')
        axes[0].set_title('Fitness Evolution')
        axes[0].legend()
        axes[0].grid(True, alpha=0.3)
        
        # Fitness improvement rate
        improvement = df_history['best_fitness'].diff()
        axes[1].bar(df_history['generation'], improvement, color='steelblue', alpha=0.7)
        axes[1].set_xlabel('Generation')
        axes[1].set_ylabel('Fitness Improvement')
        axes[1].set_title('Generation-wise Improvement')
        axes[1].grid(True, alpha=0.3)
        
        plt.suptitle('Genetic Algorithm Evolution Progress', fontsize=14)
        plt.tight_layout()
        plt.show()

GA for Feature Selection

class GAFeatureSelection(GeneticAlgorithm):
    """
    Genetic Algorithm for feature selection in ML
    """
    
    def __init__(self, X, y, estimator, cv=5, **kwargs):
        """
        Initialize GA for feature selection
        
        Parameters:
        -----------
        X : array-like
            Feature matrix
        y : array-like
            Target variable
        estimator : sklearn estimator
            ML model to evaluate feature subsets
        cv : int
            Number of cross-validation folds
        """
        self.X = X
        self.y = y
        self.estimator = estimator
        self.cv = cv
        
        # Chromosome length = number of features
        super().__init__(chromosome_length=X.shape[1], **kwargs)
        
    def fitness_function(self, individual: np.ndarray) -> float:
        """
        Evaluate feature subset using cross-validation
        """
        # Select features where gene = 1
        selected_features = np.where(individual == 1)[0]
        
        # Handle case with no features selected
        if len(selected_features) == 0:
            return 0.0
        
        # Evaluate model with selected features
        X_subset = self.X[:, selected_features]
        
        try:
            # Use cross-validation score as fitness
            scores = cross_val_score(self.estimator, X_subset, self.y, cv=self.cv)
            
            # Combine accuracy and parsimony (fewer features is better)
            accuracy = scores.mean()
            parsimony_bonus = 0.01 * (1 - len(selected_features) / len(individual))
            
            return accuracy + parsimony_bonus
            
        except:
            return 0.0
    
    def get_selected_features(self) -> np.ndarray:
        """
        Get indices of selected features
        """
        return np.where(self.best_individual == 1)[0]
    
    def print_results(self, feature_names=None):
        """
        Print feature selection results
        """
        selected_idx = self.get_selected_features()
        
        print("\n" + "=" * 50)
        print("FEATURE SELECTION RESULTS")
        print("=" * 50)
        print(f"Total features: {self.chromosome_length}")
        print(f"Selected features: {len(selected_idx)}")
        print(f"Reduction: {(1 - len(selected_idx)/self.chromosome_length)*100:.1f}%")
        print(f"Best fitness: {self.best_fitness:.4f}")
        
        if feature_names is not None:
            print("\nSelected features:")
            for idx in selected_idx:
                print(f"  - {feature_names[idx]}")

GA for Hyperparameter Optimization

class GAHyperparameterTuning:
    """
    GA for hyperparameter optimization
    """
    
    def __init__(self, param_bounds, X, y, estimator_class, cv=3):
        """
        Initialize GA for hyperparameter tuning
        
        Parameters:
        -----------
        param_bounds : dict
            Dictionary of parameter names and (min, max) bounds
        X : array-like
            Feature matrix
        y : array-like  
            Target variable
        estimator_class : class
            ML model class to optimize
        """
        self.param_bounds = param_bounds
        self.param_names = list(param_bounds.keys())
        self.X = X
        self.y = y
        self.estimator_class = estimator_class
        self.cv = cv
        
        # GA parameters
        self.population_size = 20
        self.generations = 30
        self.mutation_rate = 0.1
        self.crossover_rate = 0.8
        
        self.population = None
        self.best_params = None
        self.best_fitness = -np.inf
        self.fitness_history = []
        
    def encode_params(self, params):
        """
        Encode parameters to chromosome [0, 1] range
        """
        encoded = []
        for name in self.param_names:
            value = params[name]
            min_val, max_val = self.param_bounds[name]
            
            if isinstance(min_val, int):
                # Integer parameter
                normalized = (value - min_val) / (max_val - min_val)
            else:
                # Float parameter
                normalized = (value - min_val) / (max_val - min_val)
            
            encoded.append(normalized)
        
        return np.array(encoded)
    
    def decode_params(self, chromosome):
        """
        Decode chromosome to parameters
        """
        params = {}
        for i, name in enumerate(self.param_names):
            min_val, max_val = self.param_bounds[name]
            
            if isinstance(min_val, int):
                # Integer parameter
                value = int(chromosome[i] * (max_val - min_val) + min_val)
            else:
                # Float parameter
                value = chromosome[i] * (max_val - min_val) + min_val
            
            params[name] = value
        
        return params
    
    def fitness_function(self, chromosome):
        """
        Evaluate hyperparameters using cross-validation
        """
        # Decode chromosome to parameters
        params = self.decode_params(chromosome)
        
        try:
            # Create model with parameters
            model = self.estimator_class(**params, random_state=42)
            
            # Evaluate using cross-validation
            scores = cross_val_score(model, self.X, self.y, cv=self.cv)
            
            return scores.mean()
        except:
            return 0.0
    
    def initialize_population(self):
        """
        Initialize random population
        """
        return np.random.random((self.population_size, len(self.param_names)))
    
    def crossover(self, parent1, parent2):
        """
        Uniform crossover for real-valued encoding
        """
        if np.random.random() < self.crossover_rate:
            mask = np.random.random(len(parent1)) < 0.5
            offspring1 = np.where(mask, parent1, parent2)
            offspring2 = np.where(mask, parent2, parent1)
            return offspring1, offspring2
        return parent1.copy(), parent2.copy()
    
    def mutate(self, chromosome):
        """
        Gaussian mutation for real-valued encoding
        """
        for i in range(len(chromosome)):
            if np.random.random() < self.mutation_rate:
                # Add Gaussian noise
                chromosome[i] += np.random.normal(0, 0.1)
                # Clip to valid range
                chromosome[i] = np.clip(chromosome[i], 0, 1)
        
        return chromosome
    
    def optimize(self):
        """
        Run GA optimization
        """
        self.population = self.initialize_population()
        
        print("Starting hyperparameter optimization...")
        
        for generation in range(self.generations):
            # Evaluate fitness
            fitness_scores = np.array([self.fitness_function(ind) for ind in self.population])
            
            # Track best
            best_idx = np.argmax(fitness_scores)
            if fitness_scores[best_idx] > self.best_fitness:
                self.best_fitness = fitness_scores[best_idx]
                self.best_params = self.decode_params(self.population[best_idx])
            
            self.fitness_history.append({
                'generation': generation,
                'best_fitness': self.best_fitness,
                'avg_fitness': np.mean(fitness_scores)
            })
            
            if generation % 5 == 0:
                print(f"Generation {generation}: Best Score = {self.best_fitness:.4f}")
            
            # Create new population
            new_population = []
            
            # Elitism
            new_population.append(self.population[best_idx].copy())
            
            # Tournament selection and reproduction
            while len(new_population) < self.population_size:
                # Tournament selection
                tournament_idx = np.random.choice(self.population_size, 3)
                winner_idx = tournament_idx[np.argmax(fitness_scores[tournament_idx])]
                parent1 = self.population[winner_idx]
                
                tournament_idx = np.random.choice(self.population_size, 3)
                winner_idx = tournament_idx[np.argmax(fitness_scores[tournament_idx])]
                parent2 = self.population[winner_idx]
                
                # Crossover
                offspring1, offspring2 = self.crossover(parent1, parent2)
                
                # Mutation
                offspring1 = self.mutate(offspring1)
                offspring2 = self.mutate(offspring2)
                
                new_population.append(offspring1)
                if len(new_population) < self.population_size:
                    new_population.append(offspring2)
            
            self.population = np.array(new_population[:self.population_size])
        
        print(f"\nOptimization complete!")
        print(f"Best parameters: {self.best_params}")
        print(f"Best score: {self.best_fitness:.4f}")
        
        return self.best_params

Advanced GA Techniques

class AdvancedGA:
    """
    Advanced GA techniques
    """
    
    @staticmethod
    def adaptive_mutation(generation, max_generations, initial_rate=0.1, final_rate=0.01):
        """
        Adaptive mutation rate that decreases over time
        """
        progress = generation / max_generations
        return initial_rate * (1 - progress) + final_rate * progress
    
    @staticmethod
    def multi_point_crossover(parent1, parent2, n_points=2):
        """
        Multi-point crossover
        """
        length = len(parent1)
        
        # Generate random crossover points
        points = sorted(np.random.choice(length - 1, n_points, replace=False) + 1)
        points = [0] + points + [length]
        
        # Create offspring
        offspring1, offspring2 = parent1.copy(), parent2.copy()
        
        for i in range(0, len(points) - 1, 2):
            start, end = points[i], points[i + 1]
            offspring1[start:end] = parent2[start:end]
            offspring2[start:end] = parent1[start:end]
        
        return offspring1, offspring2
    
    @staticmethod
    def diversity_preservation(population, fitness_scores, diversity_weight=0.1):
        """
        Modify fitness scores to preserve population diversity
        """
        # Calculate pairwise distances
        n = len(population)
        distances = np.zeros((n, n))
        
        for i in range(n):
            for j in range(i + 1, n):
                # Hamming distance for binary encoding
                dist = np.sum(population[i] != population[j])
                distances[i, j] = distances[j, i] = dist
        
        # Calculate diversity score for each individual
        diversity_scores = np.mean(distances, axis=1)
        
        # Normalize diversity scores
        if diversity_scores.max() > 0:
            diversity_scores = diversity_scores / diversity_scores.max()
        
        # Combine fitness and diversity
        adjusted_fitness = fitness_scores + diversity_weight * diversity_scores
        
        return adjusted_fitness
    
    @staticmethod
    def island_model(islands, migration_rate=0.1, migration_interval=10):
        """
        Island model for parallel GA evolution
        """
        # Simulate migration between islands
        if np.random.random() < migration_rate:
            n_islands = len(islands)
            
            for i in range(n_islands):
                # Select migrant from island i
                source_island = islands[i]
                migrant_idx = np.random.randint(len(source_island))
                migrant = source_island[migrant_idx].copy()
                
                # Send to random different island
                target_island_idx = (i + np.random.randint(1, n_islands)) % n_islands
                target_island = islands[target_island_idx]
                
                # Replace random individual in target island
                replace_idx = np.random.randint(len(target_island))
                target_island[replace_idx] = migrant
        
        return islands

Best Practices

🎯 GA Best Practices

  • Population Size: Balance diversity and computation (typically 50-200)
  • Selection Pressure: Too high causes premature convergence
  • Mutation Rate: Start with 1/chromosome_length, adjust as needed
  • Elitism: Preserve best solutions (typically 5-10% of population)
  • Encoding: Choose appropriate representation for your problem
  • Fitness Function: Include penalties for constraints
  • Diversity: Monitor and maintain genetic diversity
  • Parallelization: Use island models for large problems
  • Hybrid Approaches: Combine GA with local search
  • Early Stopping: Stop if no improvement for N generations
