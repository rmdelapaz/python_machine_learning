🌐 Federated Learning: Privacy-Preserving Distributed ML

Introduction

Federated Learning enables training machine learning models on distributed datasets without centralizing the data. This approach preserves privacy by keeping data on local devices while only sharing model updates. It's crucial for applications in healthcare, finance, and mobile computing where data privacy is paramount. This lesson covers the fundamentals of federated learning, implementation strategies, and real-world applications.

Basic Federated Learning Implementation

class FederatedLearningSimulator:
    """
    Simulate federated learning with multiple clients
    """
    
    def __init__(self, n_clients=5, local_epochs=5, global_rounds=10):
        """
        Initialize federated learning system
        
        Parameters:
        -----------
        n_clients : int
            Number of federated clients
        local_epochs : int
            Training epochs per client per round
        global_rounds : int
            Number of federated rounds
        """
        self.n_clients = n_clients
        self.local_epochs = local_epochs
        self.global_rounds = global_rounds
        self.global_model = None
        self.client_models = []
        self.client_data = []
        self.history = {
            'global_accuracy': [],
            'client_accuracy': [],
            'round': []
        }
        
    def create_model(self, input_shape, n_classes):
        """
        Create neural network model
        """
        model = keras.Sequential([
            layers.Dense(64, activation='relu', input_shape=(input_shape,)),
            layers.Dropout(0.2),
            layers.Dense(32, activation='relu'),
            layers.Dropout(0.2),
            layers.Dense(n_classes, activation='softmax' if n_classes > 2 else 'sigmoid')
        ])
        
        model.compile(
            optimizer='adam',
            loss='sparse_categorical_crossentropy' if n_classes > 2 else 'binary_crossentropy',
            metrics=['accuracy']
        )
        
        return model
    
    def distribute_data(self, X, y, distribution='iid'):
        """
        Distribute data among clients
        
        Parameters:
        -----------
        distribution : str
            'iid' - Independent and Identically Distributed
            'non-iid' - Non-IID (heterogeneous)
        """
        n_samples = len(X)
        samples_per_client = n_samples // self.n_clients
        
        self.client_data = []
        
        if distribution == 'iid':
            # IID: Random distribution
            indices = np.random.permutation(n_samples)
            
            for i in range(self.n_clients):
                start_idx = i * samples_per_client
                end_idx = start_idx + samples_per_client if i < self.n_clients - 1 else n_samples
                
                client_indices = indices[start_idx:end_idx]
                self.client_data.append({
                    'X': X[client_indices],
                    'y': y[client_indices]
                })
                
        else:  # non-iid
            # Non-IID: Skewed distribution
            # Sort by labels to create heterogeneous distribution
            sorted_indices = np.argsort(y)
            
            for i in range(self.n_clients):
                # Give each client a skewed distribution
                if i < self.n_clients // 2:
                    # First half gets more of class 0
                    start_idx = i * samples_per_client // 2
                    end_idx = start_idx + samples_per_client
                else:
                    # Second half gets more of class 1
                    start_idx = n_samples - (self.n_clients - i) * samples_per_client
                    end_idx = start_idx + samples_per_client
                
                client_indices = sorted_indices[start_idx:end_idx]
                self.client_data.append({
                    'X': X[client_indices],
                    'y': y[client_indices]
                })
        
        # Print data distribution
        print(f"\nData Distribution ({distribution}):")
        for i, data in enumerate(self.client_data):
            unique, counts = np.unique(data['y'], return_counts=True)
            print(f"  Client {i+1}: {len(data['y'])} samples, "
                  f"Class distribution: {dict(zip(unique, counts))}")
    
    def federated_averaging(self, client_weights):
        """
        Federated Averaging (FedAvg) algorithm
        """
        # Get the number of layers
        n_layers = len(client_weights[0])
        
        # Initialize averaged weights
        avg_weights = []
        
        # Average weights layer by layer
        for layer_idx in range(n_layers):
            layer_weights = [client[layer_idx] for client in client_weights]
            avg_layer = np.mean(layer_weights, axis=0)
            avg_weights.append(avg_layer)
        
        return avg_weights
    
    def train_client(self, client_id, global_weights=None):
        """
        Train a single client's model
        """
        # Get client data
        X_client = self.client_data[client_id]['X']
        y_client = self.client_data[client_id]['y']
        
        # Create or update client model
        if global_weights is not None:
            # Start from global model
            client_model = self.create_model(X_client.shape[1], len(np.unique(y_client)))
            client_model.set_weights(global_weights)
        else:
            client_model = self.create_model(X_client.shape[1], len(np.unique(y_client)))
        
        # Train model
        history = client_model.fit(
            X_client, y_client,
            epochs=self.local_epochs,
            batch_size=32,
            verbose=0
        )
        
        # Evaluate
        accuracy = client_model.evaluate(X_client, y_client, verbose=0)[1]
        
        return client_model.get_weights(), accuracy
    
    def train_federated(self, X_test, y_test):
        """
        Main federated training loop
        """
        print("\n" + "=" * 60)
        print("FEDERATED TRAINING")
        print("=" * 60)
        
        # Initialize global model
        n_features = self.client_data[0]['X'].shape[1]
        n_classes = len(np.unique(self.client_data[0]['y']))
        self.global_model = self.create_model(n_features, n_classes)
        
        # Federated training rounds
        for round_num in range(self.global_rounds):
            print(f"\n--- Round {round_num + 1}/{self.global_rounds} ---")
            
            # Get current global weights
            global_weights = self.global_model.get_weights()
            
            # Train each client
            client_weights = []
            client_accuracies = []
            
            for client_id in range(self.n_clients):
                # Client training
                weights, accuracy = self.train_client(client_id, global_weights)
                client_weights.append(weights)
                client_accuracies.append(accuracy)
                print(f"  Client {client_id + 1} accuracy: {accuracy:.4f}")
            
            # Federated averaging
            averaged_weights = self.federated_averaging(client_weights)
            self.global_model.set_weights(averaged_weights)
            
            # Evaluate global model
            global_accuracy = self.global_model.evaluate(X_test, y_test, verbose=0)[1]
            
            print(f"  Global model accuracy: {global_accuracy:.4f}")
            
            # Store history
            self.history['global_accuracy'].append(global_accuracy)
            self.history['client_accuracy'].append(client_accuracies)
            self.history['round'].append(round_num + 1)
        
        return self.global_model
    
    def visualize_training(self):
        """
        Visualize federated training progress
        """
        fig, axes = plt.subplots(1, 2, figsize=(14, 5))
        
        # Global model accuracy
        axes[0].plot(self.history['round'], self.history['global_accuracy'], 
                    'o-', linewidth=2, markersize=8, color='blue')
        axes[0].set_xlabel('Federated Round')
        axes[0].set_ylabel('Accuracy')
        axes[0].set_title('Global Model Accuracy')
        axes[0].grid(True, alpha=0.3)
        
        # Client accuracies
        client_acc_array = np.array(self.history['client_accuracy'])
        for client_id in range(self.n_clients):
            axes[1].plot(self.history['round'], client_acc_array[:, client_id],
                        'o-', alpha=0.7, label=f'Client {client_id + 1}')
        
        axes[1].set_xlabel('Federated Round')
        axes[1].set_ylabel('Accuracy')
        axes[1].set_title('Client Model Accuracies')
        axes[1].legend()
        axes[1].grid(True, alpha=0.3)
        
        plt.suptitle('Federated Learning Training Progress', fontsize=14)
        plt.tight_layout()
        plt.show()

Privacy-Preserving Techniques

class PrivacyPreservingFL:
    """
    Privacy-preserving techniques for federated learning
    """
    
    def __init__(self):
        self.epsilon = 1.0  # Privacy budget
        
    def add_differential_privacy(self, weights, epsilon=1.0, delta=1e-5):
        """
        Add differential privacy noise to model weights
        
        Parameters:
        -----------
        epsilon : float
            Privacy parameter (smaller = more privacy)
        delta : float
            Failure probability
        """
        print("\n" + "=" * 60)
        print("DIFFERENTIAL PRIVACY")
        print("=" * 60)
        
        noisy_weights = []
        
        for layer_weights in weights:
            # Calculate sensitivity (L2 norm)
            sensitivity = np.linalg.norm(layer_weights)
            
            # Calculate noise scale
            noise_scale = sensitivity * np.sqrt(2 * np.log(1.25 / delta)) / epsilon
            
            # Add Gaussian noise
            noise = np.random.normal(0, noise_scale, layer_weights.shape)
            noisy_layer = layer_weights + noise
            
            noisy_weights.append(noisy_layer)
        
        print(f"Added differential privacy noise (ε={epsilon}, δ={delta})")
        print(f"Privacy guarantee: (ε, δ)-differential privacy")
        
        return noisy_weights
    
    def secure_aggregation(self, client_updates):
        """
        Simulate secure aggregation using masking
        """
        print("\n" + "=" * 60)
        print("SECURE AGGREGATION")
        print("=" * 60)
        
        n_clients = len(client_updates)
        n_layers = len(client_updates[0])
        
        # Generate random masks for each client pair
        masks = {}
        for i in range(n_clients):
            for j in range(i + 1, n_clients):
                # Shared random seed between clients i and j
                np.random.seed(i * 100 + j)
                mask = []
                for layer in client_updates[0]:
                    mask.append(np.random.randn(*layer.shape))
                masks[(i, j)] = mask
        
        # Apply masks to client updates
        masked_updates = []
        for i in range(n_clients):
            masked_update = []
            for layer_idx in range(n_layers):
                # Start with original update
                masked_layer = client_updates[i][layer_idx].copy()
                
                # Add masks for clients with smaller index
                for j in range(i):
                    mask = masks[(j, i)]
                    masked_layer += mask[layer_idx]
                
                # Subtract masks for clients with larger index
                for j in range(i + 1, n_clients):
                    mask = masks[(i, j)]
                    masked_layer -= mask[layer_idx]
                
                masked_update.append(masked_layer)
            
            masked_updates.append(masked_update)
        
        # Aggregate masked updates (masks cancel out)
        aggregated = []
        for layer_idx in range(n_layers):
            layer_sum = np.zeros_like(masked_updates[0][layer_idx])
            for client_idx in range(n_clients):
                layer_sum += masked_updates[client_idx][layer_idx]
            aggregated.append(layer_sum / n_clients)
        
        print("Secure aggregation completed")
        print("Individual updates remain private while computing average")
        
        return aggregated
    
    def homomorphic_encryption_demo(self):
        """
        Demonstrate concept of homomorphic encryption
        """
        print("\n" + "=" * 60)
        print("HOMOMORPHIC ENCRYPTION (CONCEPTUAL)")
        print("=" * 60)
        
        # Simulate homomorphic properties
        def simple_encrypt(value, key):
            """Simple encryption for demonstration"""
            return value * key + np.random.random()
        
        def simple_decrypt(encrypted, key):
            """Simple decryption for demonstration"""
            return encrypted / key
        
        # Example weights
        weight1 = 0.5
        weight2 = 0.3
        key = 7.0
        
        # Encrypt
        enc_weight1 = simple_encrypt(weight1, key)
        enc_weight2 = simple_encrypt(weight2, key)
        
        # Perform computation on encrypted values
        enc_sum = enc_weight1 + enc_weight2
        
        # Decrypt result
        decrypted_sum = simple_decrypt(enc_sum, key)
        actual_sum = weight1 + weight2
        
        print("Homomorphic Encryption Example:")
        print(f"  Original weights: {weight1:.4f}, {weight2:.4f}")
        print(f"  Encrypted weights: {enc_weight1:.4f}, {enc_weight2:.4f}")
        print(f"  Sum of encrypted: {enc_sum:.4f}")
        print(f"  Decrypted sum: {decrypted_sum:.4f}")
        print(f"  Actual sum: {actual_sum:.4f}")
        print("\nThis allows computation on encrypted data without decryption!")

class AdvancedFederatedLearning:
    """
    Advanced federated learning techniques
    """
    
    def __init__(self):
        self.strategies = {}
        
    def federated_averaging_weighted(self, client_updates, client_sizes):
        """
        Weighted federated averaging based on data size
        """
        print("\n" + "=" * 60)
        print("WEIGHTED FEDERATED AVERAGING")
        print("=" * 60)
        
        total_size = sum(client_sizes)
        weights = [size / total_size for size in client_sizes]
        
        print("Client weights based on data size:")
        for i, (size, weight) in enumerate(zip(client_sizes, weights)):
            print(f"  Client {i+1}: {size} samples, weight={weight:.3f}")
        
        # Weighted average
        n_layers = len(client_updates[0])
        weighted_avg = []
        
        for layer_idx in range(n_layers):
            layer_avg = np.zeros_like(client_updates[0][layer_idx])
            for client_idx, client_weight in enumerate(weights):
                layer_avg += client_weight * client_updates[client_idx][layer_idx]
            weighted_avg.append(layer_avg)
        
        return weighted_avg
    
    def compression_techniques(self, weights, compression_rate=0.1):
        """
        Compress model updates for efficient communication
        """
        print("\n" + "=" * 60)
        print("MODEL COMPRESSION")
        print("=" * 60)
        
        compressed_weights = []
        original_size = 0
        compressed_size = 0
        
        for layer_weights in weights:
            original_size += layer_weights.size
            
            # Method 1: Top-k sparsification
            flat_weights = layer_weights.flatten()
            k = int(len(flat_weights) * compression_rate)
            
            # Get top k values by magnitude
            top_k_indices = np.argpartition(np.abs(flat_weights), -k)[-k:]
            
            # Create sparse representation
            sparse_weights = np.zeros_like(flat_weights)
            sparse_weights[top_k_indices] = flat_weights[top_k_indices]
            
            # Reshape back
            compressed_layer = sparse_weights.reshape(layer_weights.shape)
            compressed_weights.append(compressed_layer)
            
            # Count non-zero elements
            compressed_size += np.count_nonzero(compressed_layer)
        
        compression_ratio = 1 - (compressed_size / original_size)
        
        print(f"Compression Statistics:")
        print(f"  Original size: {original_size} parameters")
        print(f"  Compressed size: {compressed_size} parameters")
        print(f"  Compression ratio: {compression_ratio:.2%}")
        print(f"  Bandwidth savings: {compression_ratio:.2%}")
        
        return compressed_weights
    
    def personalized_federated_learning(self, global_model, client_data, alpha=0.5):
        """
        Personalized FL: Blend global and local models
        """
        print("\n" + "=" * 60)
        print("PERSONALIZED FEDERATED LEARNING")
        print("=" * 60)
        
        # Simulate personalization for each client
        personalized_models = []
        
        print(f"Personalization factor α={alpha}")
        print("Personalized = α * Local + (1-α) * Global\n")
        
        for client_id, data in enumerate(client_data):
            # Simulate local model weights
            local_weights = [layer + np.random.randn(*layer.shape) * 0.1 
                           for layer in global_model]
            
            # Blend with global model
            personalized = []
            for local, global_layer in zip(local_weights, global_model):
                blended = alpha * local + (1 - alpha) * global_layer
                personalized.append(blended)
            
            personalized_models.append(personalized)
            
            # Calculate similarity to global model
            similarity = np.mean([np.corrcoef(p.flatten(), g.flatten())[0, 1]
                                 for p, g in zip(personalized, global_model)])
            
            print(f"Client {client_id + 1}: Similarity to global = {similarity:.3f}")
        
        return personalized_models
    
    def detect_byzantine_clients(self, client_updates, threshold=2.0):
        """
        Detect potentially malicious or faulty clients
        """
        print("\n" + "=" * 60)
        print("BYZANTINE FAULT DETECTION")
        print("=" * 60)
        
        n_clients = len(client_updates)
        
        # Calculate median update
        median_update = []
        for layer_idx in range(len(client_updates[0])):
            layer_stack = np.array([client[layer_idx] for client in client_updates])
            median_layer = np.median(layer_stack, axis=0)
            median_update.append(median_layer)
        
        # Calculate distance from median for each client
        distances = []
        for client_idx, client_update in enumerate(client_updates):
            # L2 distance from median
            dist = sum([np.linalg.norm(client_layer - median_layer)
                       for client_layer, median_layer in zip(client_update, median_update)])
            distances.append(dist)
        
        # Detect outliers using z-score
        mean_dist = np.mean(distances)
        std_dist = np.std(distances)
        z_scores = [(d - mean_dist) / std_dist for d in distances]
        
        byzantine_clients = []
        for client_idx, z_score in enumerate(z_scores):
            if abs(z_score) > threshold:
                byzantine_clients.append(client_idx)
                print(f"  ⚠️ Client {client_idx + 1}: Potential Byzantine (z={z_score:.2f})")
            else:
                print(f"  ✓ Client {client_idx + 1}: Normal (z={z_score:.2f})")
        
        print(f"\nDetected {len(byzantine_clients)} potential Byzantine clients")
        
        return byzantine_clients

Best Practices

🎯 Federated Learning Best Practices

  • Client Selection: Randomly sample clients to avoid bias
  • Communication Efficiency: Use compression and quantization
  • Handling Non-IID Data: Use personalization or clustering
  • Privacy Guarantees: Implement differential privacy
  • Secure Aggregation: Protect individual updates
  • Byzantine Robustness: Detect and handle malicious clients
  • System Heterogeneity: Account for different device capabilities
  • Dropout Handling: Manage clients that disconnect
  • Model Convergence: Monitor and ensure convergence
  • Fairness: Ensure model works well for all participants
